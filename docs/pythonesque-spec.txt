def _generate_historical_ordinal(self) -> str:
number = self._rng.randint(1, 14)
suffixes = {1: "st", 2: "nd", 3: "rd"}
suffix = "th" if 11 <= number <= 13 else suffixes.get(number % 10, "th")
return f"{number}{suffix}"
def _build_proper_noun(self) -> str:
title = self._rng.choice(self._grammar["name_title"])
first = self._rng.choice(self._grammar["name_first"])
last_1 = self._rng.choice(self._grammar["name_last"])
if self._rng.random() < 0.15:
first = f"{first} 'The {self._rng.choice(self._grammar['pejorative'])}'"
if self._rng.random() < 0.30:
last_2 = self._rng.choice(self._grammar["name_last"])
while last_2 == last_1:
last_2 = self._rng.choice(self._grammar["name_last"])
surname = f"{last_1}-{last_2}"
else:
surname = last_1
return f"{title} {first} {surname}" if title else f"{first} {surname}"
def _build_enhanced_institution(self) -> str:
base_inst = self._rng.choice(self._grammar["institution_base"])
desc = self._rng.choice(self._grammar["institution_descriptor"])
return f"{desc} {base_inst}" if desc else base_inst
def _clean_whitespace(self, text: str) -> str:
text = re.sub(r'\s+', ' ', text)
return text.strip()
def _fix_indefinite_articles(self, text: str) -> str:
def replace_match(match):
article = match.group(1)
following_word = match.group(2)
if following_word.lower() in 'aeiou':
return "An " if article.isupper() or article.startswith('A') else "an "
return match.group(0)
return re.sub(r'\b([Aa])\s+(\w+)', replace_match, text)
def _resolve(self, token: str) -> str:
if token == "historical_ordinal":
return self._generate_historical_ordinal()
if token == "generated_proper_noun":
return self._build_proper_noun()
if token == "enhanced_institution":
return self._build_enhanced_institution()
if token == "generated_legal_action":
core = self._resolve(self._rng.choice(self._grammar["legal_core_action"]))
pattern = self._resolve(self._rng.choice(self._grammar["legal_situational_pattern"]))
return f"{core} {pattern}" if pattern else core
if token == "generated_natural_philosophy_axiom":
prefix = self._rng.choice(self._grammar["science_prefix"])
prop = self._rng.choice(self._grammar["science_property"])
conn = self._rng.choice(self._grammar["science_connector"])
target = self._resolve(self._rng.choice(self._grammar["science_target"]))
constraint = self._rng.choice(self._grammar["science_constraint"])
core_vector = f"{prefix} {prop}" if prefix else prop
return f"{core_vector} {conn} {target} {constraint}" if constraint else f"{core_vector} {conn} {target}"
if token == "generated_academic_body":
if self._rng.random() < 0.20:
return self._rng.choice(self._grammar["academic_body_noun"])
mod = self._rng.choice(self._grammar["academic_body_modifier"])
noun = self._rng.choice(self._grammar["academic_body_noun"])
return f"{mod} {noun}"
if token == "generated_logistics_action":
if self._rng.random() < 0.30:
return self._rng.choice(self._grammar["logistics_vector"])
tier = self._rng.choice(self._grammar["logistics_tier"])
vector = self._rng.choice(self._grammar["logistics_vector"])
return f"{tier} {vector}"
if token == "generated_academic_discipline":
prefix = self._rng.choice(self._grammar["academic_prefix"])
subject = self._resolve("academic_subject_structure")
return f"{prefix} {subject}" if prefix else subject
if token == "noun_verbing":
return f"{self._rng.choice(self._grammar['verbing_base_noun'])}-{self._rng.choice(self._grammar['verbing_suffix_gerund'])}"
if token not in self._grammar:
return token
chosen_pattern = self._rng.choice(self._grammar[token])
pattern_finder = re.compile(r"{{(\w+)}}")
while True:
match = pattern_finder.search(chosen_pattern)
if not match:
break
nested_token = match.group(1)
replacement_value = self._resolve(nested_token)
chosen_pattern = chosen_pattern.replace(match.group(0), replacement_value, 1)
return chosen_pattern
def generate(self, slugify: bool = False) -> str:
phrase = self._resolve("START")
phrase = self._clean_whitespace(phrase)
phrase = self._fix_indefinite_articles(phrase)
if slugify:
clean_str = re.sub(r"[^a-zA-Z0-9\s-]", "", phrase)
return "-".join(clean_str.lower().split())
return phrase
